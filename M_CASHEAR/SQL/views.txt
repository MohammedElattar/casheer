create view hagz_view as
select h.H_NO,h.H_NAME,h.H_DAY,h.H_FROM,h.H_TO,h.TEL_NO,h.JOB,
k.ROOM_NO,k.ESKAN_TYP,h.out_date,h.out_flag
from hagz_tbl h,hagz_eskan k
where   h.H_NO = k.H_NO
group by h.H_NO,h.H_NAME,h.H_DAY,h.H_FROM,h.H_TO,h.TEL_NO,h.JOB,
         k.ROOM_NO,k.ESKAN_TYP,h.out_date,h.out_flag;

create view dtl_view as
select o.SERIAL_NO,o.POINT_CODE,o.INV_DATE,o.EMP_ID,o.SHIFT_TYP,sum(nvl(I_Q,0)) s 
from invoice_hist o,inv_dtl_hist d
where
o.SERIAL_NO = d.SERIAL_NO and o.POINT_CODE = d.POINT_CODE and
o.INV_DATE = d.INV_DATE and o.EMP_ID = d.EMP_ID and
o.SHIFT_TYP = d.SHIFT_TYP and o.p_ok = 1
group by o.SERIAL_NO,o.POINT_CODE,o.INV_DATE,o.EMP_ID,o.SHIFT_TYP;
