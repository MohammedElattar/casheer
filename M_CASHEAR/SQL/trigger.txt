cursor cur is
select H_NO,ROOM_NO,EKAMA_TYP,CUST_TYP,PRICE from hagz_room where h_no = :old.h_no;

if :new.out_date is not null then
   delete from hagz_eskan where h_no = :old.h_no;
   delete from hagz_eskan_hist where h_no = :old.h_no;
   delete from room_account where h_no = :old.h_no and r_date > :new.out_date;
   update room_account set price = 0 where h_no = :old.h_no and r_date >= :new.out_date;

else
   delete from hagz_eskan where h_no = :old.h_no;
   delete from hagz_eskan_hist where h_no = :old.h_no;
   d := :new.h_to;flag := 0;
   delete from room_account where h_no = :old.h_no;
end if;
k := d - :new.h_from;t:=0;
------------------------------------------------
for i in cur loop
    t:=0;
    while t <= k loop
          insert into hagz_eskan_hist(H_NO,ROOM_NO,EKAMA_DATE,ESKAN_TYP,CUST_NAME,
                                 PRICE,EKAMA_TYP,CUST_TYP)
                 values(:old.H_NO,i.ROOM_NO,d - t,:new.h_flag,:new.h_name,i.PRICE,
                        i.EKAMA_TYP,i.CUST_TYP);
      if flag = 0 then
         insert into hagz_eskan(H_NO,ROOM_NO,EKAMA_DATE,ESKAN_TYP,CUST_NAME,
                                 PRICE,EKAMA_TYP,CUST_TYP)
                 values(:old.H_NO,i.ROOM_NO,d - t,:new.h_flag,:new.h_name,i.PRICE,
                        i.EKAMA_TYP,i.CUST_TYP);

         insert into room_account(H_NO,ROOM_NO,r_DATE,price)
                 values(:new.H_NO,i.ROOM_NO,d - t,i.PRICE);
      end if;
          t := t + 1;
    end loop;
------------------------------

end loop;
END;

declare
c number;h_d number(5);h_f date;h_t date;h_n varchar2(100);
k number;t number;
BEGIN
 select h_day,h_from,h_to,h_name into h_d,h_f,h_t,h_n from hagz_tbl 
  where h_no = :new.h_no; 
k := h_t - h_f; 
t := 0;
while t <= k loop
  insert into hagz_eskan(H_NO,ROOM_NO,EKAMA_DATE,ESKAN_TYP,CUST_NAME,PRICE,EKAMA_TYP,CUST_TYP)
    values(:new.H_NO,:new.ROOM_NO,h_t - t,1,h_n,:new.PRICE,:new.EKAMA_TYP,:new.CUST_TYP);
  insert into hagz_eskan_hist(H_NO,ROOM_NO,EKAMA_DATE,ESKAN_TYP,CUST_NAME,PRICE,EKAMA_TYP,CUST_TYP)
    values(:new.H_NO,:new.ROOM_NO,h_t - t,1,h_n,:new.PRICE,:new.EKAMA_TYP,:new.CUST_TYP);

  insert into room_account(H_NO,ROOM_NO,r_DATE,price)
    values(:new.H_NO,:new.ROOM_NO,h_t - t,:new.PRICE);
  t := t + 1;
end loop;   
END;
