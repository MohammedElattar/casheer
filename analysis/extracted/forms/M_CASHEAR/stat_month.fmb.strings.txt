BUTTON_PROC
-- This is a generic button procedure. It reads the NAME of the button and 
-- performs a DO_KEY(item_name).
-- Regarding QUERY-operation:
-- if there are buttons called EXECUTE_QUERY and CANCEL_QUERY, this function shows
-- them, when the ENTER_QUERY button is pressed and hides them, when EXECUTE_- or 
-- CANCEL_QUERY ispressed. No error should be returned, if these buttons do not exist.
-- To make naming of the buttons easier, EXIT, QUIT and EXIT_FORM all perform exit_form,
-- even if the form is in ENTER-QUERY mode !!!!!!!
-- A CANCEL_QUERY button-name cancels a currently "open" query.
PROCEDURE button_proc IS
  AL_ID    ALERT;
  AL_BUT   NUMBER;
  action varchar(80);
  -- hide a button (for ENTER_-, EXECUTE_- or CANCEL_QUERY)
  PROCEDURE show_off(item_name VARCHAR2) IS
  BEGIN
    IF NOT Id_Null(Find_Item(item_name)) THEN
      Set_Item_Property(item_name, DISPLAYED,PROPERTY_FALSE);
    END IF;
  END;
  -- try to show a button (for the two query-states. If there is no button to show,
  -- return FALSE to the calling procedure.
  FUNCTION show_on(item_name VARCHAR2) RETURN BOOLEAN IS
  BEGIN
    IF NOT Id_Null(Find_Item(item_name)) THEN
      Set_Item_Property(item_name,DISPLAYED,PROPERTY_TRUE);
      Set_Item_Property(item_name,ENABLED,PROPERTY_TRUE);
      RETURN TRUE;
    ELSE
      RETURN FALSE;
    END IF;
  RETURN NULL; END;
BEGIN    
  action := Get_Item_Property(NAME_IN('SYSTEM.TRIGGER_ITEM'),ITEM_NAME);
  IF action = 'ENTER_QUERY' THEN
    -- only hide the ENTER_QUERY button, if there are EXECUTE_QUERY and (!!) CANCEL_QUERY
    -- buttons. Otherwise an existing EXECUTE_QUERY button will be shown.
    IF show_on('EXECUTE_QUERY') AND show_on('CANCEL_QUERY') THEN
      show_off('ENTER_QUERY');
    END IF;
  END IF;
  IF action IN ('EXIT_FORM', 'EXIT','QUIT') THEN
    action := 'EXIT_FORM';
    IF NAME_IN('SYSTEM.MODE')='ENTER-QUERY' THEN
           AL_ID:=FIND_ALERT('MESS_AL');
           Set_Alert_Property(al_id, alert_message_text, '        
           al_but:=show_alert(al_id); 
              IF  al_but =ALERT_BUTTON1 THEN
                 Do_Key('EXIT_FORM');
              ELSE 
                  NULL;
              END IF;
    END IF;
  END IF;
  IF action = 'CANCEL_QUERY' THEN 
    action := 'EXIT_FORM';
  END IF;
  Do_Key(action);
  IF NAME_IN('SYSTEM.MODE') != 'ENTER-QUERY' THEN
    IF show_on('ENTER_QUERY') THEN
      show_off('EXECUTE_QUERY');
      show_off('CANCEL_QUERY');
    END IF;
  END IF;

  BEGIN

      Delete_Timer('ENTER_TOOLBAR');
    END IF;

    t Timer;

                       to_number(Get_Item_Property(item_name,X_POS)),
                       to_number(Get_Item_Property(item_name,Y_POS)) +
                       to_number(Get_Item_Property(item_name,HEIGHT)));
    Set_Item_Property('TOOLBAR.TEXT',DISPLAYED,PROPERTY_ON);

  cs := Get_Form_Property(Get_Application_Property(CURRENT_FORM_NAME),COORDINATE_SYSTEM);

   PROCEDURE run
      display     IN VARCHAR2, 
      item        IN VARCHAR2,
      refresh     IN BOOLEAN,
      plist       IN TOOLS.paramlist
   BEGIN
      -- don't shut down display
      -- the following should be commented out for windows 3.1
      --TOOLS.add_parameter (plist, 'oracle_shutdown', 
      --                     TOOLS.text_parameter, 'no');
      -- pass back vgs display list
      IF (refresh) THEN
         TOOLS.add_parameter(plist,
                             'oracle_update',
                             TOOLS.text_parameter, 
                             'yes');
      ELSE
         TOOLS.add_parameter(plist,
                             'oracle_update',
                             TOOLS.text_parameter, 
                             'no');
      END IF;
      -- do not call startup trigger if already loaded
      TOOLS.add_parameter (plist, 
                           'oracle_startup', 
                           TOOLS.text_parameter, 
                           'no');
      TOOLS.run_product (TOOLS.graphics, 
                         display, 
                         TOOLS.synchronous, 
                         TOOLS.batch, 
                         TOOLS.filesystem, 
                         plist, 
                         item);
   END run;
   PROCEDURE mouseupdown
      display      IN VARCHAR2, 
      item         IN VARCHAR2,
      x            IN NUMBER, 
      y            IN NUMBER, 
      refresh      IN BOOLEAN,
      clickcount   IN NUMBER,
      button       IN NUMBER,
      constrained  IN BOOLEAN,
      plist        IN TOOLS.paramlist,
      down         IN BOOLEAN
      pl     TOOLS.paramlist;
      xtmp   NUMBER;
      ytmp   NUMBER;
      xscale NUMBER;
      yscale NUMBER;
   BEGIN
      IF (coord_system = coords_real) THEN
         IF (units = units_inch) THEN
            xscale := 1;
            yscale := 1;
         ELSIF (units = units_cm) THEN
            xscale := 2.540;
            yscale := 2.540;
         ELSIF (units = units_pica) THEN
            xscale := 72;
            yscale := 72;
         ELSE
            xscale := resolution;
            yscale := resolution;
         END IF;
      ELSE
         xscale := resolution / cell_width;
         yscale := resolution / cell_height;
      END IF;
      IF (x = 0) THEN
         xtmp := TO_NUMBER (NAME_IN('system.mouse_x_pos')) / xscale * 8192;
      END IF;
      IF (y = 0) THEN
         ytmp := TO_NUMBER (NAME_IN('system.mouse_y_pos')) / yscale * 8192;
      END IF;
      --  create parameter list to pass mouse up event to embedded display.
      IF (NOT (TOOLS.isnull (plist))) THEN
         pl := plist;
      ELSE
         pl := TOOLS.create_parameter_list ('oracle_olelist');
      END IF;
      -- pass mouse coordinates
      IF (down) THEN
         BEGIN
            TOOLS.add_parameter (pl, 
                                 'oracle_mousedown', 
                                 TOOLS.text_parameter, 
                                 TO_CHAR (TRUNC(xtmp)) || ' ' || 
                                 TO_CHAR (TRUNC(ytmp)) || ' ' ||
                                                       '1' || ' ' || '1');
         END;
      ELSE
         BEGIN
            TOOLS.add_parameter (pl, 
                                 'oracle_mouseup', 
            TOOLS.text_parameter, 
            TO_CHAR (TRUNC(xtmp)) || ' ' || TO_CHAR (TRUNC(ytmp)) ||
                     ' ' || '1' || ' ' || '1');
         END;
      END IF;
      run (display,item, refresh, pl);
      IF (TOOLS.isnull (plist)) THEN
         TOOLS.destroy_parameter_list ('oracle_olelist');
      END IF;
   END mouseupdown;
   PROCEDURE open
      display      IN VARCHAR2, 
      item         IN VARCHAR2,
      clip         IN BOOLEAN := TRUE,
      refresh      IN BOOLEAN := TRUE,
      plist        IN TOOLS.paramlist := TOOLS.null_parameter_list
      pl        TOOLS.paramlist;
      x         NUMBER  :=
      y         NUMBER  := 0; 
      width     NUMBER  := 0;
      height    NUMBER  := 0;
      dispmode  NUMBER  := 0;
      fstr      VARCHAR(255);
      ustr      VARCHAR(255);
   BEGIN
      fstr := NAME_IN('system.current_form');
      ustr := GET_FORM_PROPERTY (fstr, COORDINATE_SYSTEM);
      coord_system := coords_real;
      IF (ustr = 'INCHES') THEN
         units := units_inch;
      ELSIF (ustr = 'CENTIMETERS') THEN
         units := units_cm;
      ELSIF (ustr = 'PIXELS') THEN
         units := units_pixel;
      ELSIF (ustr = 'POINTS') THEN
         units := units_pica;
      ELSE
         BEGIN
            coord_system := coords_char;
            ustr := GET_FORM_PROPERTY (fstr, CHARACTER_CELL_WIDTH);
            cell_width := TO_NUMBER(ustr);
            ustr := GET_FORM_PROPERTY (fstr, CHARACTER_CELL_HEIGHT);
            cell_height := TO_NUMBER(ustr);
         END;
      END IF;
      IF (clip) THEN
         dispmode := 1;
      END IF;
      --  create parameter list to pass mouse up event to embedded display.
      IF (NOT (TOOLS.isnull (plist))) THEN
         pl := plist;
      ELSE
         pl := TOOLS.create_parameter_list ('oracle_olelist');
      END IF;
      -- pass mouse coordinates
      TOOLS.add_parameter (pl, 'oracle_viewport', TOOLS.text_parameter,
         TO_CHAR (TRUNC(x)) || ' ' || TO_CHAR (TRUNC(y)) || ' ' ||
         TO_CHAR(TRUNC(width)) || ' ' || TO_CHAR(TRUNC(height)) || ' ' ||
         TO_CHAR(dispmode));
      run (display, item, refresh, pl);
      IF (TOOLS.isnull (plist)) THEN
         TOOLS.destroy_parameter_list ('oracle_olelist');
      END IF;
   END open;

   END close;
