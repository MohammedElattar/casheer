--- PU #2 name=P_EXIT kind=1 ---
PROCEDURE P_EXIT 
IS
	vALertId  		ALERT;
	vAlertButton  NUMBER;	
BEGIN
	-- Check On Form Change Done and Save Button is Enabled
  IF NAME_IN('SYSTEM.FORM_STATUS') = 'CHANGED' 
  	 AND GET_ITEM_PROPERTY('TOOLBAR.SAVE',ENABLED) = 'TRUE' THEN	
	   vALertId := FIND_ALERT('THREEBUTTON');
	   IF NOT ID_NULL(vALertId) THEN
	   	  -- Set Alert Text ( Title,Message,Button(s) )
        SET_ALERT_PROPERTY(vALertId,TITLE,'Choices/?IE?C?CE');
        SET_ALERT_PROPERTY(vALertId,ALERT_MESSAGE_TEXT,'Save Changes/??U C?E?I?? ?');
        SET_ALERT_BUTTON_PROPERTY(vALertId,ALERT_BUTTON1,LABEL,'Yes/???');
        SET_ALERT_BUTTON_PROPERTY(vALertId,ALERT_BUTTON2,LABEL,'No/?C');
        SET_ALERT_BUTTON_PROPERTY(vALertId,ALERT_BUTTON3,LABEL,'Cancel/???C?');

        -- Get Return Value
    	  vAlertButton := SHOW_ALERT(vALertId);

    	  
    	  -- Perform Action Needed per return value
	      IF    vAlertButton = ALERT_BUTTON1 THEN
	   	        DO_KEY('COMMIT_FORM');
	      ELSIF vAlertButton = ALERT_BUTTON2 THEN
         	 EXIT_FORM(NO_VALIDATE);   
	      ELSIF vAlertButton = ALERT_BUTTON3 THEN
           NULL;	      	
	      END IF;	   	    	   
	      
	   END IF;
	 ELSE  
	    EXIT_FORM(NO_VALIDATE);   
   END IF;
END;
